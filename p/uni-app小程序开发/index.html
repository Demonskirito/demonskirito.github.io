  • calculateDate(_date):

    • 获取给定日期所在周的日期范围（从周日到周六）。

    • 更新日期数组，将计算出的日期加入到 dateList 中，决定是否将其插入到数组前面或后面（取决于 atBefore 参数）。

稍微解释一下：

  • booking 表：存储预约的基本信息，如预约ID、预约日期、开始和结束时间，以及剩余可预约次数。

  • booking_records 表：存储具体的预约记录，包含场馆ID、用户ID、预约状态（已完成、已取消或未完成）、预约时间等信息。

  • venue_schedule 表：存储每个场馆的可预约时间，包含开放时间和关闭时间。

  • venues 表：存储场馆的基本信息，包括场馆名称、地点、状态（是否可预约）以及场馆的图片链接。

现在的要求（幸好我是全栈，还能理清思路）

前端会给一个string的current_time,在booking中寻找对应的current_time,并根据start_time对应open_time,end_time对应close_time,将remaining的值与参 left join 到venue_schedule表中，只需调用api接口就可以知道每个日期对应的预约数目 如果没有对应的current_time和正确对应上open_time=start_time,close_time=end_time,remaining默认left join时为3

SELECT 
    vs.schedule_id,
    vs.venue_id,
    vs.open_time, 
    vs.close_time, 
    IFNULL(b.remaining, 3) AS remaining,  -- 如果没有预约数据，remaining 默认值为 3
    COUNT(br.booking_id) AS booked_count
FROM 
    venue_schedule vs
LEFT JOIN 
    booking b 
    ON b.start_time >= vs.open_time 
    AND b.end_time <= vs.close_time
    AND b.current_time = "2025-1-19"  -- 根据传入的 current_time 筛选
LEFT JOIN 
    booking_records br 
    ON b.id = br.booking_id 
    AND br.booking_status = 0  -- 预约状态为已完成
GROUP BY 
    vs.venue_id, vs.open_time, vs.close_time;

export default {
  data() {
    return {
      isLoggedIn: false,
      openid: '' // 存储 openid
    };
  },
  methods: {
    // 登录操作
    goToLogin() {
      console.log("enter goToLogin");

      if (!this.isLoggedIn) {
        // 使用 uni.getUserProfile 获取用户信息
        uni.getUserProfile({
          desc: '获取你的昵称、头像等信息',  // 提示用户需要获取的权限
          success: (res_user) => {
            console.log(res_user.userInfo);  // 获取到的用户信息

            // 获取用户的登录 code
            uni.login({
              success: (res) => {
                if (res.code) {
                  console.log(res.code); // 获取到的 code

                  // 发起网络请求
                  uni.request({
                    url: "http://localhost:8081/user/login", // 后端 URL
                    method: "POST",
                    data: {
                      code: res.code,
                      imgUrl: res_user.userInfo.avatarUrl,
                      nickname: res_user.userInfo.nickName,
                      sex: res_user.userInfo.gender,
                    },
                    header: {
                      "Content-Type": "application/x-www-form-urlencoded",
                    },
                    success: (response) => {
                      if (response.statusCode === 200) {
                        console.log("登录成功，openid:", response.data.openid);
                        uni.setStorageSync("openid", response.data.openid);
                        this.openid = response.data.openid; // 更新 openid
                        this.isLoggedIn = true; // 标记用户为已登录
                      } else {
                        console.log("登录失败", response.errMsg);
                      }
                    },
                    fail: (err) => {
                      console.log("网络请求失败", err);
                    },
                  });
                } else {
                  console.log("获取用户登录态失败！" + res.errMsg);
                }
              },
            });
          },
          fail: (err) => {
            console.log("用户拒绝授权或发生错误", err);
          }
        });
      } else {
        uni.showToast({
          title: "已登录",
          icon: "success",
        });
      }
    },
  }
};
</script>

public  String GET(String url) {  
    String result="";  
    BufferedReader in = null;  
    InputStream is = null;  
    InputStreamReader isr = null;  
    try{  
        //构造url对象  
        URL realUrl = new URL(url);  
        //获取一个对应该URL的URLConnection对象  
        URLConnection conn = realUrl.openConnection();  
        conn.connect();  
        is = conn.getInputStream();  
        isr = new InputStreamReader(is);  
        in = new BufferedReader(isr);  
        String line;  
        while ((line = in.readLine()) != null) {  
            result += line;  
        }  
    }catch (Exception e){  
        e.printStackTrace();  
        System.out.println("异常出现");  
    }  
    //关闭资源  
    finally {  
        try {  
            if (in != null) {  
                in.close();  
            }  
            if (is != null) {  
                is.close();  
            }  
            if (isr != null) {  
                isr.close();  
            }  
        } catch (Exception e) {  
            e.printStackTrace();  
            System.out.println("出现异常");  
        }  
    }  
    return result;  
}

package com.example.teachhealth.config;  
  
public class ApiResponse<T> {  
    private int code;          // 状态码  
    private String message;    // 提示信息  
    private T data;            // 响应数据  
  
    // 全参构造器  
    public ApiResponse(int code, String message, T data) {  
        this.code = code;  
        this.message = message;  
        this.data = data;  
    }  
  
    // 静态方法构造成功和失败的响应  
    public static <T> ApiResponse<T> success(T data) {  
        return new ApiResponse<>(200, "操作成功", data);  
    }  
  
    public static <T> ApiResponse<T> success() {  
        return new ApiResponse<>(200, "操作成功", null);  
    }  
  
    public static <T> ApiResponse<T> failure(String message) {  
        return new ApiResponse<>(400, message, null);  
    }  
  
    public static <T> ApiResponse<T> failure(int code, String message) {  
        return new ApiResponse<>(code, message, null);  
    }  
  
    // Getters 和 Setters    public int getCode() {  
        return code;  
    }  
  
    public void setCode(int code) {  
        this.code = code;  
    }  
  
    public String getMessage() {  
        return message;  
    }  
  
    public void setMessage(String message) {  
        this.message = message;  
    }  
  
    public T getData() {  
        return data;  
    }  
  
    public void setData(T data) {  
        this.data = data;  
    }  
}

public ApiResponse<?> add(HttpServletRequest request, @RequestBody BookingRecords bookingRecords) {  
    Enumeration<String> headerNames = request.getHeaderNames();  
    while (headerNames.hasMoreElements()) {  
        String headerName = headerNames.nextElement();  
        System.out.println(headerName + ": " + request.getHeader(headerName));  
    }  
  
  
    String openid = bookingRecords.getOpenid();  
    String currentDay = bookingRecords.getCurrentDay();  
    Integer venueId = bookingRecords.getVenueId();  
    Time startTime = bookingRecords.getStartTime();  
    Time endTime = bookingRecords.getEndTime();  
  
    System.out.println(currentDay + " " + venueId + " " + startTime + " " + endTime);  
  
    // 判断是否存在相同的预约记录  
    if (bookingRecordsService.isBookingExist(venueId, openid, currentDay, startTime, endTime)) {  
        System.out.println("预约失败：该时间段已被占用！");  
        return ApiResponse.failure("预约失败：该时间段已被占用！");  
    }  
  
    // 添加或更新预约记录  
    bookingService.addOrUpdateBooking(venueId, currentDay, startTime, endTime);  
    boolean result = bookingRecordsService.save(bookingRecords);  
  
    if (result) {  
        System.out.println("预约成功！");  
        return ApiResponse.success("预约成功！");  
    } else {  
        System.out.println("预约失败：保存记录时出现问题！");  
        return ApiResponse.failure("预约失败：保存记录时出现问题！");  
    }  
}

function buildCommentTree(comments) {
  const commentMap = {}; // 存储评论ID到评论对象的映射
  const rootComments = []; // 存储主评论（无父评论）

  // 初始化所有评论，并添加children属性
  comments.forEach(comment => {
    commentMap[comment.id] = { ...comment, children: [] };
  });

  // 构建树结构
  comments.forEach(comment => {
    const parentId = comment.parentCommentId;
    if (parentId === null) {
      rootComments.push(commentMap[comment.id]); // 添加到主评论
    } else {
      const parent = commentMap[parentId];
      if (parent) {
        parent.children.push(commentMap[comment.id]); // 添加到父评论的children
      } else {
        console.warn(`父评论 ${parentId} 不存在，无法关联到评论 ${comment.id}`);
      }
    }
  });

  return rootComments;
}

  comments.forEach(comment => {

pojo vo dto区别

POJO (Plain Old Java Object)

  • 定义：POJO 是一个普通的 Java 对象，不依赖于任何特定的框架或库。
  • 特点：
    • 没有继承或实现特定的接口。
    • 主要用于存储数据。
    • 通常包含私有字段和公共的 getter 和 setter 方法。

VO (Value Object)

  • 定义：VO 是一种特定类型的 POJO，通常用于表示业务层中的数据。
  • 特点：
    • 通常是不可变的（字段是 final 的）。
    • 用于传递只读数据。
    • 主要用于业务逻辑层。

DTO (Data Transfer Object)

  • 定义：DTO 是一种用于在不同层之间传输数据的对象，通常用于服务层和控制层之间。
  • 特点：
    • 主要用于数据传输。
    • 可以包含验证逻辑。
    • 通常包含与数据库表对应的字段。

总结

  • POJO 是最基础的 Java 对象，用于存储数据。
  • VO 是一种特定的 POJO，用于表示业务层中的数据，通常是不可变的。
  • DTO 用于在不同层之间传输数据，通常包含与数据库表对应的字段。
